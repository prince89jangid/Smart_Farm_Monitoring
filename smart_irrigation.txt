/*
 * ============================================================================
 *  GreenDrip - ESP32 Smart Irrigation Controller
 * ============================================================================
 *  Hardware
 *    Soil moisture sensor (analog) : AO  -> GPIO 34  (ADC1, input only)
 *                                    VCC -> 3V3,  GND -> GND
 *    Relay module                  : IN  -> GPIO 26
 *                                    VCC -> 5V (or 3V3, check your module), GND -> GND
 *                                    Pump wires through relay COM + NO
 *    Buzzer (active type)          : (+) -> GPIO 27,  (-) -> GND
 *
 *  Logic
 *    1. Soil moisture is read continuously and shown on the web dashboard.
 *    2. If moisture <= LOW threshold (Automatic mode): buzzer beeps
 *       "beep-beep" and the relay (pump) turns ON.
 *    3. Pump turns OFF when moisture >= HIGH threshold, or after the max-run
 *       safety time. Buzzer gives one long beep when the pump stops.
 *    4. Thresholds can be changed from the dashboard (saved in flash).
 *
 *  Dashboard (built in - nothing else to upload)
 *    The full web dashboard is embedded in this file. After uploading:
 *      1. Open Serial Monitor (115200) and note the IP address.
 *      2. On a phone/PC connected to the same Wi-Fi open  http://<that IP>
 *         (or http://esp32.local).
 *    If the ESP32 cannot join your Wi-Fi it starts its own hotspot
 *    "GreenDrip-ESP32" (password 12345678) -> open http://192.168.4.1
 *
 *  Calibration (no re-flashing needed) - Serial Monitor, 115200 baud:
 *      d  = save the CURRENT reading as "dry"   (sensor in open air)
 *      w  = save the CURRENT reading as "wet"   (sensor in a glass of water)
 *      r  = reset calibration to defaults,   ?  = help
 *    Do this once, then the moisture % and the thresholds match your sensor.
 *
 *  Libraries: only the ESP32 Arduino core (WiFi, WebServer, ESPmDNS,
 *  Preferences). No extra libraries needed.
 * ============================================================================
 */

#include <WiFi.h>
#include <WebServer.h>
#include <ESPmDNS.h>
#include <Preferences.h>
#include <sys/time.h>
#include <time.h>
#include <Wire.h>
#include <hd44780.h>
#include <hd44780ioClass/hd44780_I2Cexp.h>


void handleRoot();  // serves the embedded dashboard (defined at the bottom)

// ----------------------------- USER CONFIG ---------------------------------
const char* WIFI_SSID = "Redmi";
const char* WIFI_PASS = "987654321";

#define SOIL_PIN 34
#define RELAY_PIN 23
#define BUZZER_PIN 26

#define RELAY_ACTIVE_LOW true    // most 1-channel relay modules switch ON with LOW
#define BUZZER_ACTIVE_HIGH true  // active buzzer: HIGH = sound

// Sensor calibration (raw ADC 0..4095). Measure your own sensor:
//   ADC_DRY = reading with sensor in open air (dry)
//   ADC_WET = reading with sensor in a glass of water
// Capacitive sensors read HIGH when dry and LOW when wet.
int ADC_DRY = 3200;
int ADC_WET = 1300;

const float PUMP_FLOW_L_PER_MIN = 1.5;  // used only to estimate water volume
const long GMT_OFFSET_SEC = 19800;      // India (UTC+5:30)
const char* FIRMWARE_VERSION = "v1.1.0";
const uint32_t CONFIRM_MS = 2000;  // a threshold must hold this long before the pump switches (ignores spikes)

// ------------------------------ SETTINGS -----------------------------------
struct Settings {
  float low = 35;       // start watering at or below (%)
  float high = 70;      // stop watering at or above (%)
  int maxRunMin = 10;   // safety cut-off (minutes)
  int cooldownMin = 2;  // rest time between automatic cycles (minutes)
} cfg;

bool autoMode = true;  // true = automatic, false = manual
Preferences prefs;
WebServer server(80);

// ------------------------------ STATE --------------------------------------
float moisture = 0;  // smoothed percentage
int rawValue = 0;    // smoothed raw ADC
bool pumpOn = false;
uint32_t sessStartMs = 0;
uint64_t sessStartEpoch = 0;
float sessStartMoist = 0;
bool sessAuto = false;
uint32_t cooldownUntilMs = 0;
bool cooldownActive = false;
bool moistureInit = false;
uint32_t lowSinceMs = 0, highSinceMs = 0;  // debounce timers for auto start / stop
bool lowTimerOn = false, highTimerOn = false;

char lastStopReason[16] = "";
uint64_t lastStopEpoch = 0;

// Moisture history: 1 sample / minute, last 24 h
#define HIST_SIZE 1440
struct HistPoint {
  uint64_t t;
  float v;
};
HistPoint history[HIST_SIZE];
int histCount = 0, histHead = 0;
uint32_t lastHistMs = 0;

// Finished irrigation sessions: last 20
#define SESS_SIZE 20
struct Session {
  uint64_t start, end;
  float durationSec, startMoist, endMoist, waterL;
  bool isAuto;
  char reason[16];
};
Session sessions[SESS_SIZE];
int sessCount = 0, sessHead = 0;

// ------------------------------ BUZZER -------------------------------------
// Non-blocking beep sequencer
uint8_t beepsLeft = 0;
uint16_t beepOnMs = 120, beepOffMs = 120;
bool beepState = false;
uint32_t beepNextMs = 0;

void buzzerWrite(bool on) {
  digitalWrite(BUZZER_PIN, (on == BUZZER_ACTIVE_HIGH) ? HIGH : LOW);
}
void startBeeps(uint8_t count, uint16_t onMs, uint16_t offMs) {
  beepsLeft = count;
  beepOnMs = onMs;
  beepOffMs = offMs;
  beepState = false;
  beepNextMs = millis();
}
void updateBuzzer() {
  if (beepsLeft == 0 && !beepState) return;
  if ((int32_t)(millis() - beepNextMs) < 0) return;
  if (!beepState) {  // start a beep
    buzzerWrite(true);
    beepState = true;
    beepNextMs = millis() + beepOnMs;
  } else {  // end a beep
    buzzerWrite(false);
    beepState = false;
    beepNextMs = millis() + beepOffMs;
    if (beepsLeft > 0) beepsLeft--;
  }
}


// ------------------------------ LCD 16x2 (I2C) -----------------------------
hd44780_I2Cexp lcd;  // I2C address auto-detect
bool lcdOk = false;

void lcdLine(uint8_t row, const char* s) {
  char buf[17];
  snprintf(buf, sizeof(buf), "%-16s", s);  // pad to 16 chars so old text is erased
  lcd.setCursor(0, row);
  lcd.print(buf);
}

void lcdInit() {
  Wire.begin(21, 22);  // SDA = 21, SCL = 22
  lcdOk = (lcd.begin(16, 2) == 0);
  if (!lcdOk) {
    Serial.println("LCD init failed - check wiring");
    return;
  }
  lcd.backlight();
  lcd.clear();
  lcdLine(0, "GreenDrip");
  lcdLine(1, "Starting...");
}

void updateLcd() {
  if (!lcdOk) return;
  char l1[24], l2[24];
  snprintf(l1, sizeof(l1), "Moisture: %5.1f%%", moisture);
  snprintf(l2, sizeof(l2), "Pump:%-3s %s", pumpOn ? "ON" : "OFF", autoMode ? "AUTO" : "MANUAL");
  lcdLine(0, l1);
  lcdLine(1, l2);
}

// ------------------------------ HELPERS ------------------------------------
uint64_t nowEpochMs() {
  struct timeval tv;
  gettimeofday(&tv, nullptr);
  return (uint64_t)tv.tv_sec * 1000ULL + tv.tv_usec / 1000;
}
bool timeSynced() {
  return time(nullptr) > 1700000000;
}

void relayWrite(bool on) {
  digitalWrite(RELAY_PIN, (on == !RELAY_ACTIVE_LOW) ? HIGH : LOW);
}

void loadSettings() {
  prefs.begin("greendrip", true);
  cfg.low = prefs.getFloat("low", 35);
  cfg.high = prefs.getFloat("high", 70);
  cfg.maxRunMin = prefs.getInt("maxRun", 10);
  cfg.cooldownMin = prefs.getInt("cool", 2);
  autoMode = prefs.getBool("auto", true);
  ADC_DRY = prefs.getInt("adcDry", ADC_DRY);
  ADC_WET = prefs.getInt("adcWet", ADC_WET);
  prefs.end();
}
void saveSettings() {
  prefs.begin("greendrip", false);
  prefs.putFloat("low", cfg.low);
  prefs.putFloat("high", cfg.high);
  prefs.putInt("maxRun", cfg.maxRunMin);
  prefs.putInt("cool", cfg.cooldownMin);
  prefs.putBool("auto", autoMode);
  prefs.end();
}
void saveCalibration() {
  prefs.begin("greendrip", false);
  prefs.putInt("adcDry", ADC_DRY);
  prefs.putInt("adcWet", ADC_WET);
  prefs.end();
}

// ------------------------------ SENSOR -------------------------------------
void readSensor() {
  long sum = 0;
  for (int i = 0; i < 16; i++) {
    sum += analogRead(SOIL_PIN);
    delayMicroseconds(200);
  }
  rawValue = sum / 16;
  int span = ADC_DRY - ADC_WET;
  float pct = (abs(span) < 50) ? 0.0f : (float)(ADC_DRY - rawValue) * 100.0f / (float)span;
  pct = constrain(pct, 0.0f, 100.0f);
  if (!moistureInit) {
    moisture = pct;
    moistureInit = true;
  } else moisture = moisture * 0.7f + pct * 0.3f;  // smooth
}

// ------------------------------ PUMP CONTROL -------------------------------
float sessionElapsedSec() {
  return pumpOn ? (millis() - sessStartMs) / 1000.0f : 0;
}

// Remaining cool-down in ms (0 = none). Uses a flag, so it stays correct
// when millis() wraps / after many days of uptime.
uint32_t cooldownLeftMs() {
  if (!cooldownActive) return 0;
  int32_t d = (int32_t)(cooldownUntilMs - millis());
  if (d <= 0) {
    cooldownActive = false;
    return 0;
  }
  return (uint32_t)d;
}

void startPump(bool automatic) {
  if (pumpOn) return;
  pumpOn = true;
  sessAuto = automatic;
  lowTimerOn = highTimerOn = false;
  sessStartMs = millis();
  sessStartEpoch = nowEpochMs();
  sessStartMoist = moisture;
  relayWrite(true);
  startBeeps(2, 150, 150);  // beep-beep
  Serial.printf("[PUMP] ON (%s) moisture=%.1f%%\n", automatic ? "auto" : "manual", moisture);
}

void stopPump(const char* reason) {
  if (!pumpOn) return;
  float dur = sessionElapsedSec();
  Session& s = sessions[sessHead];
  s.start = sessStartEpoch;
  s.end = nowEpochMs();
  s.durationSec = dur;
  s.startMoist = sessStartMoist;
  s.endMoist = moisture;
  s.waterL = dur * PUMP_FLOW_L_PER_MIN / 60.0f;
  s.isAuto = sessAuto;
  strncpy(s.reason, reason, sizeof(s.reason) - 1);
  s.reason[sizeof(s.reason) - 1] = 0;
  sessHead = (sessHead + 1) % SESS_SIZE;
  if (sessCount < SESS_SIZE) sessCount++;

  pumpOn = false;
  relayWrite(false);
  lowTimerOn = highTimerOn = false;
  strncpy(lastStopReason, reason, sizeof(lastStopReason) - 1);
  lastStopEpoch = nowEpochMs();
  cooldownActive = (strcmp(reason, "manual") != 0) && cfg.cooldownMin > 0;
  cooldownUntilMs = millis() + (uint32_t)cfg.cooldownMin * 60000UL;
  if (strcmp(reason, "max_runtime") == 0 || strcmp(reason, "overwater") == 0)
    startBeeps(3, 200, 150);   // alarm: stopped by a safety limit
  else startBeeps(1, 600, 0);  // one long beep
  Serial.printf("[PUMP] OFF (%s) after %.0fs, moisture=%.1f%%\n", reason, dur, moisture);
}

// true once `cond` has been continuously true for CONFIRM_MS
bool confirmed(bool cond, bool& on, uint32_t& since) {
  if (!cond) {
    on = false;
    return false;
  }
  if (!on) {
    on = true;
    since = millis();
    return false;
  }
  return (uint32_t)(millis() - since) >= CONFIRM_MS;
}

void controlLoop() {
  if (pumpOn) {
    // safety limits - always active, in every mode
    if (sessionElapsedSec() >= cfg.maxRunMin * 60.0f) {
      stopPump("max_runtime");
      return;
    }
    if (moisture >= 95) {
      stopPump("overwater");
      return;
    }
    lowTimerOn = false;
    if (confirmed(autoMode && moisture >= cfg.high, highTimerOn, highSinceMs)) stopPump("target_reached");
  } else {
    highTimerOn = false;
    if (confirmed(autoMode && moisture <= cfg.low && cooldownLeftMs() == 0, lowTimerOn, lowSinceMs)) startPump(true);
  }
}

void recordHistory() {
  if (millis() - lastHistMs < 60000UL && histCount > 0) return;
  if (!timeSynced()) return;
  lastHistMs = millis();
  history[histHead] = { nowEpochMs(), moisture };
  histHead = (histHead + 1) % HIST_SIZE;
  if (histCount < HIST_SIZE) histCount++;
}

// ------------------------------ HTTP API -----------------------------------
void cors() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET,POST,OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "Content-Type");
  server.sendHeader("Access-Control-Allow-Private-Network", "true");
  server.sendHeader("Cache-Control", "no-store");
}
void sendJson(int code, const String& body) {
  cors();
  server.send(code, "application/json", body);
}

String u64(uint64_t v) {
  char b[24];
  snprintf(b, sizeof(b), "%llu", (unsigned long long)v);
  return String(b);
}

bool jsonNum(const String& b, const char* key, float& out) {
  int i = b.indexOf(String("\"") + key + "\"");
  if (i < 0) return false;
  i = b.indexOf(':', i);
  if (i < 0) return false;
  out = b.substring(i + 1).toFloat();
  return true;
}
bool jsonTrue(const String& b, const char* key) {
  int i = b.indexOf(String("\"") + key + "\"");
  if (i < 0) return false;
  i = b.indexOf(':', i);
  if (i < 0) return false;
  String v = b.substring(i + 1);
  v.trim();
  return v.startsWith("true");
}

void handleStatus() {
  float elapsed = sessionElapsedSec();
  long cool = (long)((cooldownLeftMs() + 999) / 1000);
  String j = "{";
  j += "\"ts\":" + u64(nowEpochMs());
  j += ",\"moisture\":" + String(moisture, 1);
  j += ",\"raw\":" + String(rawValue);
  j += String(",\"pump\":") + (pumpOn ? "true" : "false");
  j += String(",\"mode\":\"") + (autoMode ? "auto" : "manual") + "\"";
  j += ",\"cooldownRemainingSec\":" + String(cool);
  if (pumpOn) {
    j += ",\"session\":{\"active\":true,\"startedAt\":" + u64(sessStartEpoch);
    j += ",\"startMoisture\":" + String(sessStartMoist, 1);
    j += ",\"elapsedSec\":" + String(elapsed, 0);
    j += ",\"waterL\":" + String(elapsed * PUMP_FLOW_L_PER_MIN / 60.0f, 2);
    j += String(",\"trigger\":\"") + (sessAuto ? "auto" : "manual") + "\"}";
  } else {
    j += ",\"session\":{\"active\":false}";
  }
  if (lastStopReason[0]) j += String(",\"lastStop\":{\"reason\":\"") + lastStopReason + "\",\"ts\":" + u64(lastStopEpoch) + "}";
  else j += ",\"lastStop\":null";
  j += ",\"device\":{\"ip\":\"" + WiFi.localIP().toString() + "\"";
  j += ",\"rssi\":" + String(WiFi.RSSI());
  j += String(",\"firmware\":\"") + FIRMWARE_VERSION + "\"";
  j += ",\"uptimeSec\":" + String(millis() / 1000) + "}}";
  sendJson(200, j);
}

String settingsJson() {
  return "{\"low\":" + String(cfg.low, 0) + ",\"high\":" + String(cfg.high, 0) + ",\"maxRunMin\":" + String(cfg.maxRunMin) + ",\"cooldownMin\":" + String(cfg.cooldownMin) + "}";
}
void handleGetSettings() {
  sendJson(200, settingsJson());
}

void handlePostSettings() {
  String b = server.arg("plain");
  float v;
  Settings n = cfg;
  if (jsonNum(b, "low", v)) n.low = v;
  if (jsonNum(b, "high", v)) n.high = v;
  if (jsonNum(b, "maxRunMin", v)) n.maxRunMin = (int)v;
  if (jsonNum(b, "cooldownMin", v)) n.cooldownMin = (int)v;
  if (n.low < 0 || n.high > 100 || n.high < n.low + 10 || n.maxRunMin < 1 || n.cooldownMin < 0) {
    sendJson(400, "{\"error\":\"invalid thresholds (high must be at least low + 10)\"}");
    return;
  }
  cfg = n;
  saveSettings();
  Serial.printf("[CFG] low=%.0f high=%.0f maxRun=%d cool=%d\n", cfg.low, cfg.high, cfg.maxRunMin, cfg.cooldownMin);
  sendJson(200, settingsJson());
}

void handlePump() {
  bool on = jsonTrue(server.arg("plain"), "on");
  if (on) startPump(false);
  else stopPump("manual");
  sendJson(200, "{\"ok\":true}");
}
void handleMode() {
  autoMode = server.arg("plain").indexOf("auto") >= 0;
  saveSettings();
  Serial.printf("[MODE] %s\n", autoMode ? "auto" : "manual");
  sendJson(200, "{\"ok\":true}");
}

void handleHistory() {
  float rangeMs = 86400000.0f;
  if (server.hasArg("range")) rangeMs = server.arg("range").toFloat();
  uint64_t now = nowEpochMs(), from = (rangeMs >= now) ? 0 : now - (uint64_t)rangeMs;
  server.setContentLength(CONTENT_LENGTH_UNKNOWN);
  cors();
  server.send(200, "application/json", "");
  String chunk = "[";
  bool first = true;
  int start = (histCount < HIST_SIZE) ? 0 : histHead;
  for (int i = 0; i < histCount; i++) {
    HistPoint& p = history[(start + i) % HIST_SIZE];
    if (p.t < from) continue;
    if (!first) chunk += ",";
    chunk += "{\"t\":" + u64(p.t) + ",\"v\":" + String(p.v, 1) + "}";
    first = false;
    if (chunk.length() > 900) {
      server.sendContent(chunk);
      chunk = "";
    }
  }
  chunk += "]";
  server.sendContent(chunk);
  server.sendContent("");
}

void handleSessions() {
  int limit = server.hasArg("limit") ? server.arg("limit").toInt() : 20;
  if (limit < 1 || limit > SESS_SIZE) limit = SESS_SIZE;
  String j = "[";
  int out = 0;
  for (int i = 0; i < sessCount && out < limit; i++) {  // newest first
    Session& s = sessions[(sessHead - 1 - i + SESS_SIZE) % SESS_SIZE];
    if (out++) j += ",";
    j += "{\"id\":" + u64(s.start) + ",\"start\":" + u64(s.start) + ",\"end\":" + u64(s.end);
    j += ",\"durationSec\":" + String(s.durationSec, 0);
    j += ",\"startMoisture\":" + String(s.startMoist, 1) + ",\"endMoisture\":" + String(s.endMoist, 1);
    j += ",\"waterL\":" + String(s.waterL, 2);
    j += String(",\"trigger\":\"") + (s.isAuto ? "auto" : "manual") + "\",\"reason\":\"" + s.reason + "\"}";
  }
  j += "]";
  sendJson(200, j);
}

void handleNotFound() {
  if (server.method() == HTTP_OPTIONS) {
    cors();
    server.send(204);
    return;
  }  // CORS pre-flight
  sendJson(404, "{\"error\":\"not found\"}");
}

// ------------------------------ SERIAL CALIBRATION -------------------------
void printHelp() {
  Serial.println("--- Serial commands ---");
  Serial.println("d = save current reading as DRY (sensor in air)");
  Serial.println("w = save current reading as WET (sensor in water)");
  Serial.println("r = reset calibration, ? = this help");
  Serial.printf("Current calibration: DRY=%d  WET=%d\n", ADC_DRY, ADC_WET);
}
void handleSerial() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'd' || c == 'D') {
      ADC_DRY = rawValue;
      saveCalibration();
      moistureInit = false;
      Serial.printf("[CAL] DRY saved = %d\n", ADC_DRY);
    } else if (c == 'w' || c == 'W') {
      ADC_WET = rawValue;
      saveCalibration();
      moistureInit = false;
      Serial.printf("[CAL] WET saved = %d\n", ADC_WET);
    } else if (c == 'r' || c == 'R') {
      ADC_DRY = 3200;
      ADC_WET = 1300;
      saveCalibration();
      moistureInit = false;
      Serial.println("[CAL] reset to defaults");
    } else if (c == '?' || c == 'h' || c == 'H') printHelp();
  }
}

// ------------------------------ SETUP / LOOP -------------------------------
void connectWifi() {
  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.print("Connecting to WiFi");
  uint32_t t0 = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t0 < 20000) {
    delay(400);
    Serial.print(".");
  }
  if (WiFi.status() == WL_CONNECTED) Serial.printf("\nConnected. IP: http://%s\n", WiFi.localIP().toString().c_str());
  else {
    Serial.println("\nWiFi failed - starting hotspot 'GreenDrip-ESP32' (password 12345678).");
    Serial.println("Connect to it and open http://192.168.4.1");
    WiFi.mode(WIFI_AP_STA);
    WiFi.softAP("GreenDrip-ESP32", "12345678");
  }
  if (lcdOk) {
    lcdLine(0, "WiFi connected");
    lcdLine(1, WiFi.localIP().toString().c_str());
    delay(1500);
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  relayWrite(false);  // pump OFF at boot
  pinMode(BUZZER_PIN, OUTPUT);
  buzzerWrite(false);
  analogReadResolution(12);
  analogSetPinAttenuation(SOIL_PIN, ADC_11db);  // 0 - ~3.3 V range

  loadSettings();
  readSensor();  // first reading


  lcdInit();  // <-- add this line


  connectWifi();
  configTime(GMT_OFFSET_SEC, 0, "pool.ntp.org", "time.google.com");
  if (MDNS.begin("esp32")) Serial.println("mDNS: http://esp32.local");

  server.on("/", HTTP_GET, handleRoot);
  server.on("/favicon.ico", HTTP_GET, []() {
    server.send(204);
  });
  server.on("/api/status", HTTP_GET, handleStatus);
  server.on("/api/settings", HTTP_GET, handleGetSettings);
  server.on("/api/settings", HTTP_POST, handlePostSettings);
  server.on("/api/pump", HTTP_POST, handlePump);
  server.on("/api/mode", HTTP_POST, handleMode);
  server.on("/api/history", HTTP_GET, handleHistory);
  server.on("/api/sessions", HTTP_GET, handleSessions);
  server.onNotFound(handleNotFound);
  server.begin();

  startBeeps(1, 80, 0);  // short "ready" chirp
  Serial.println("GreenDrip ready. Open the dashboard in your browser.");
  printHelp();
}

void loop() {
  server.handleClient();
  handleSerial();
  updateBuzzer();

  static uint32_t lastRead = 0;
  if (millis() - lastRead >= 500) {
    lastRead = millis();
    readSensor();
    controlLoop();
    recordHistory();
    updateLcd();
    static uint32_t lastPrint = 0;
    if (millis() - lastPrint >= 2000) {
      lastPrint = millis();
      Serial.printf("raw=%d  moisture=%.1f%%  pump=%s  mode=%s  (start<=%.0f stop>=%.0f)  cooldown=%lus\n",
                    rawValue, moisture, pumpOn ? "ON" : "OFF", autoMode ? "AUTO" : "MANUAL", cfg.low, cfg.high,
                    (unsigned long)((cooldownLeftMs() + 999) / 1000));
    }
  }
}

class DemoDevice {
  constructor(){
    this.LIVE_EVAP = 0.05;                 // %/s of simulated drying
    this.speed = 1; this.offline = false;
    this.bootAt = Date.now() - 3*86400e3 - 4*3600e3;
    this.reset(false);

  }
  reset(clear=true){

    this.settings = Object.assign({}, DEFAULT_SETTINGS, store.get('demo.settings', {}) || {});
    this.mode = store.get('demo.mode', null) || 'auto';
    this.pump = false; this.session = null; this.sessions = []; this.history = [];
    this.cooldownLeft = 0; this.lastStop = null; this.lastRec = 0; this.m = 58;
    this.seed();
  }
  seed(){                                   // builds 24 h of realistic history using the same control logic
    const end = Date.now(), keepMode = this.mode; this.mode = 'auto';
    let t = end - 24*3600e3; this.m = 62; const sp = this.speed; this.speed = 1;
    while(t < end){
      const d = new Date(t), hr = d.getHours() + d.getMinutes()/60;
      this.evap = 0.0012 + 0.0024*Math.max(0, Math.sin((hr-6)/12*Math.PI));
      this.step(5, t);
      if(t - this.lastRec >= ((end-t) > 15*60e3 ? 60e3 : 5e3)){ this.history.push({t, v:+this.m.toFixed(2)}); this.lastRec = t; }
      t += 5000;
    }
    this.mode = keepMode; this.speed = sp; this.evap = this.LIVE_EVAP;
    this.lastTick = Date.now(); this.lastRec = end;
  }
  startPump(trigger, now){
    this.pump = true;
    this.session = { id: now, startedAt: now, startMoisture: this.m, elapsedSec: 0, waterL: 0, trigger };
  }

    const s = this.session; if(!s){ this.pump = false; return; }
    this.sessions.push({ id:s.id, start:s.startedAt, end:now, durationSec:s.elapsedSec, startMoisture:s.startMoisture, endMoisture:this.m, waterL:s.waterL, trigger:s.trigger, reason });
    if(this.sessions.length > 200) this.sessions.shift();
    this.pump = false; this.session = null; this.lastStop = { reason, ts: now };
    this.cooldownLeft = reason==='manual' ? 0 : this.settings.cooldownMin*60;
  }
  step(dt, now){                            // dt = simulated seconds
    const s = this.settings;
    if(this.cooldownLeft > 0) this.cooldownLeft = Math.max(0, this.cooldownLeft - dt);
    if(this.pump){
      this.session.elapsedSec += dt;
      if(this.session.elapsedSec >= s.maxRunMin*60) this.stopPump('max_runtime', now);
      else if(this.m >= 95) this.stopPump('overwater', now);
      else if(this.mode==='auto' && this.m >= s.high) this.stopPump('target_reached', now);
    } else if(this.mode==='auto' && this.m <= s.low && this.cooldownLeft <= 0){
      this.startPump('auto', now);
    }
    if(this.pump){ this.m += 1.3*(1 - this.m/100)*dt; this.session.waterL += 0.025*dt; }
    else this.m -= this.evap*dt;
    this.m = clamp(this.m + (Math.random()-.5)*0.05*Math.sqrt(dt), 0, 100);
  }
  advance(){
    const now = Date.now(); let sim = Math.min((now - this.lastTick)/1000, 600) * this.speed; this.lastTick = now;
    while(sim > 0){ const d = Math.min(sim, 2); this.step(d, now); sim -= d; }
    if(now - this.lastRec >= 5000){
      this.history.push({ t:now, v:+this.m.toFixed(2) }); this.lastRec = now;
      while(this.history.length && this.history[0].t < now - 25*3600e3) this.history.shift();
    }
  }
  snapshot(){                               // identical shape to what the real firmware should return
    const now = Date.now();
    return {
      ts: now, moisture: +this.m.toFixed(1), raw: Math.round(3200 - this.m*16.5), pump: this.pump, mode: this.mode,
      cooldownRemainingSec: Math.ceil(this.cooldownLeft),
      session: this.session ? { active:true, ...this.session } : { active:false },
      lastStop: this.lastStop,
      device: { ip:'192.168.1.42', rssi: -58 + Math.round(Math.sin(now/9000)*5), firmware:'v1.4.2-demo', uptimeSec:(now-this.bootAt)/1000 }
    };
  }
}

function downsample(pts, t0, t1, n){
  if(pts.length <= n*1.5) return pts;
  const out = [], bw = (t1-t0)/n; let i = 0;
  for(let b=0; b<n; b++){
    const bs = t0+b*bw, be = bs+bw; let sum=0, c=0, ts=0;
    while(i < pts.length && pts[i].t < be){ if(pts[i].t >= bs){ sum+=pts[i].v; ts+=pts[i].t; c++; } i++; }
    if(c) out.push({ t:ts/c, v:sum/c });
  }
  const last = pts[pts.length-1]; if(out.length && out[out.length-1].t < last.t) out.push(last);
  return out;
}

    if(cv.width !== Math.round(w*dpr) || cv.height !== Math.round(h*dpr)){ cv.width = Math.round(w*dpr); cv.height = Math.round(h*dpr); }
    const ctx = this.ctx; ctx.setTransform(dpr,0,0,dpr,0,0); ctx.clearRect(0,0,w,h);
    const col = this.colors(), pad = { l:36, r:12, t:12, b:26 }, pw = w-pad.l-pad.r, ph = h-pad.t-pad.b;
    const now = Date.now(), range = S.range, t0 = now-range;
    const X = t => pad.l + (t-t0)/range*pw, Y = v => pad.t + (1-v/100)*ph, cfg = S.settings;
    ctx.font = `600 11px ${col.font}`; ctx.textBaseline = 'middle';

    // target band
    ctx.fillStyle = hexA(col.accent, .12); ctx.fillRect(pad.l, Y(cfg.high), pw, Y(cfg.low)-Y(cfg.high));
    // pump-on bands
    for(const b of this.bands(now)){
      const x1 = X(Math.max(b.a,t0)), x2 = X(Math.min(b.b,now)); if(x2 <= pad.l || x2 < x1) continue;
      ctx.fillStyle = hexA(col.water, .16); ctx.fillRect(x1, pad.t, Math.max(2,x2-x1), ph);
      ctx.fillStyle = hexA(col.water, .85); ctx.fillRect(x1, pad.t+ph-4, Math.max(2,x2-x1), 4);
    }
    // grid + y labels
    ctx.textAlign = 'right'; ctx.lineWidth = 1;
    for(const v of [0,25,50,75,100]){
      ctx.strokeStyle = hexA(col.muted, v===0 ? .35 : .16); ctx.beginPath(); ctx.moveTo(pad.l, Y(v)+.5); ctx.lineTo(w-pad.r, Y(v)+.5); ctx.stroke();
      ctx.fillStyle = col.muted; ctx.fillText(v+'%', pad.l-7, Y(v));
    }
    // thresholds
    ctx.setLineDash([5,4]); ctx.strokeStyle = hexA(col.accent, .7); ctx.lineWidth = 1.2;
    for(const v of [cfg.low, cfg.high]){ ctx.beginPath(); ctx.moveTo(pad.l, Y(v)); ctx.lineTo(w-pad.r, Y(v)); ctx.stroke(); }
    ctx.setLineDash([]);
    // x ticks
    const target = Math.max(3, Math.floor(pw/95)), ideal = range/target;
    const steps = [1,2,5,10,15,30,60,120,180,360,720].map(m=>m*60000), step = steps.find(s=>s>=ideal) || steps[steps.length-1];
    const off = new Date().getTimezoneOffset()*60000; ctx.textAlign = 'center';
    for(let t = Math.ceil((t0-off)/step)*step+off; t <= now; t += step){
      const x = X(t); if(x < pad.l+14 || x > w-pad.r-14) continue;
      ctx.strokeStyle = hexA(col.muted, .12); ctx.beginPath(); ctx.moveTo(x+.5, pad.t); ctx.lineTo(x+.5, pad.t+ph); ctx.stroke();
      ctx.fillStyle = col.muted; ctx.fillText(fmtTime(t), x, h-10);
    }
    // line + area
    let pts = S.history.filter(p => p.t >= t0 - range*0.03);
    pts = downsample(pts, t0, now, Math.max(40, Math.floor(pw/2.5)));
    const vis = pts.map(p => ({ x:X(p.t), y:Y(p.v), t:p.t, v:p.v })); this.view = { pad, vis, t0, w, h, ph, bands:this.bands(now) };
    if(vis.length > 1){
      ctx.beginPath(); vis.forEach((p,i) => i ? ctx.lineTo(p.x,p.y) : ctx.moveTo(p.x,p.y));
      const g = ctx.createLinearGradient(0, pad.t, 0, pad.t+ph); g.addColorStop(0, hexA(col.accent,.32)); g.addColorStop(1, hexA(col.accent,0));
      ctx.save(); ctx.lineTo(vis[vis.length-1].x, pad.t+ph); ctx.lineTo(vis[0].x, pad.t+ph); ctx.closePath(); ctx.fillStyle = g; ctx.fill(); ctx.restore();
      ctx.beginPath(); vis.forEach((p,i) => i ? ctx.lineTo(p.x,p.y) : ctx.moveTo(p.x,p.y));
      ctx.lineWidth = 2.6; ctx.lineJoin = 'round'; ctx.lineCap = 'round'; ctx.strokeStyle = col.accent; ctx.stroke();
      const l = vis[vis.length-1];
      ctx.fillStyle = hexA(col.accent,.25); ctx.beginPath(); ctx.arc(l.x,l.y,8,0,7); ctx.fill();
      ctx.fillStyle = col.accent; ctx.strokeStyle = col.surface; ctx.lineWidth = 2.5; ctx.beginPath(); ctx.arc(l.x,l.y,4.5,0,7); ctx.fill(); ctx.stroke();
    } else if(!vis.length){
      ctx.fillStyle = col.muted; ctx.textAlign = 'center'; ctx.font = `600 13px ${col.font}`; ctx.fillText('No data for this range yet', pad.l+pw/2, pad.t+ph/2);
    }
    // hover
    if(this.hover){
      const p = this.hover;
      ctx.strokeStyle = hexA(col.text,.35); ctx.lineWidth = 1; ctx.setLineDash([3,3]); ctx.beginPath(); ctx.moveTo(p.x+.5, pad.t); ctx.lineTo(p.x+.5, pad.t+ph); ctx.stroke(); ctx.setLineDash([]);
      ctx.fillStyle = col.surface; ctx.strokeStyle = col.accent; ctx.lineWidth = 3; ctx.beginPath(); ctx.arc(p.x,p.y,5.5,0,7); ctx.fill(); ctx.stroke();
    }
  }
  onMove(e){
    if(!this.view || !this.view.vis.length) return;
    const r = this.c.getBoundingClientRect(), x = e.clientX - r.left, vis = this.view.vis;
    let lo = 0, hi = vis.length-1;
    while(hi-lo > 1){ const mid = (lo+hi)>>1; (vis[mid].x < x) ? lo = mid : hi = mid; }
    const p = Math.abs(vis[lo].x-x) < Math.abs(vis[hi].x-x) ? vis[lo] : vis[hi];
    if(Math.abs(p.x-x) > 40){ this.hover = null; this.tip.hidden = true; this.draw(); return; }
    this.hover = p; const pumping = this.view.bands.some(b => p.t >= b.a && p.t <= b.b);
    this.tip.hidden = false;
    this.tip.innerHTML = `<b>${p.v.toFixed(1)}%</b><br>${fmtTime(p.t)}${pumping ? ' · 💧 pump on' : ''}`;
    const tw = this.tip.offsetWidth/2 + 4;
    this.tip.style.left = clamp(p.x, tw, this.view.w - tw) + 'px'; this.tip.style.top = Math.max(34, p.y - 12) + 'px';
    this.draw();
  }
}

function rate(){

}

function renderProcess(st, cfg){
  const ses = st.session, ls = st.lastStop, recent = !!(ls && !ses && st.ts - ls.ts < 45000);
  const safety = recent && (ls.reason === 'max_runtime' || ls.reason === 'overwater');
  let phase = 'idle', label = 'Idle';
  if(ses){ phase = 'watering'; label = 'Watering'; }
  else if(safety){ phase = 'cutoff'; label = 'Safety stop'; }
  else if(st.mode === 'auto' && st.cooldownRemainingSec > 0){ phase = 'cooldown'; label = 'Cooldown'; }
  const ph = $('#phase'); ph.dataset.phase = phase; ph.textContent = label;

  // stepper
  const lis = $$('#steps li'); let states = ['','','',''];
  if(ses) states = ['done','done','active',''];
  else if(recent) states = ['done','done','done', safety ? 'warn' : 'done'];
  lis.forEach((li,i) => li.className = states[i]);
  lis[0].querySelector('.s-label').textContent = ses ? (ses.trigger === 'auto' ? 'Dry detected' : 'Manual start') : 'Soil check';
  lis[3].querySelector('.s-label').textContent = safety ? 'Safety stop' : recent && ls.reason === 'manual' ? 'Stopped' : 'Complete';

  // progress
  let pct = 0;
  if(ses){ const span = cfg.high - ses.startMoisture; pct = span <= 0 ? 100 : clamp((st.moisture - ses.startMoisture)/span*100, 0, 100); }
  else if(recent && !safety && ls.reason === 'target_reached') pct = 100;
  $('#progLabel').textContent = `Progress to target (${cfg.high}%)`; $('#progPct').textContent = Math.round(pct)+'%';
  $('#barFill').style.width = pct+'%'; $('#bar').classList.toggle('live', !!ses); $('#bar').setAttribute('aria-valuenow', Math.round(pct));

  // stats
  const last = S.sessions[0];
  if(ses){
    const gain = st.moisture - ses.startMoisture, rt = ses.elapsedSec > 6 ? gain/ses.elapsedSec : 0;
    const eta = rt > .02 && st.moisture < cfg.high ? (cfg.high - st.moisture)/rt : null;
    $('#statCap').textContent = 'Current session';
    $('#stDur').textContent = fmtDur(ses.elapsedSec); $('#stDurS').textContent = `max ${fmtDur(cfg.maxRunMin*60)}`;
    $('#stWater').textContent = ses.waterL.toFixed(2)+' L'; $('#stWaterS').textContent = `≈ ${(ses.waterL/Math.max(ses.elapsedSec,1)*60).toFixed(1)} L/min`;
    $('#stGain').textContent = (gain>=0?'+':'')+gain.toFixed(1)+'%'; $('#stGainS').textContent = `from ${ses.startMoisture.toFixed(1)}%`;
    $('#stEtaK').textContent = 'Time to target'; $('#stEta').textContent = eta != null ? '~'+fmtDur(eta) : '—'; $('#stEtaS').textContent = `to reach ${cfg.high}%`;
  } else if(last){
    const gain = last.endMoisture - last.startMoisture;
    $('#statCap').textContent = 'Last session';
    $('#stDur').textContent = fmtDur(last.durationSec); $('#stDurS').textContent = last.trigger === 'auto' ? 'automatic' : 'manual';
    $('#stWater').textContent = last.waterL.toFixed(2)+' L'; $('#stWaterS').textContent = 'delivered';
    $('#stGain').textContent = (gain>=0?'+':'')+gain.toFixed(1)+'%'; $('#stGainS').textContent = `${last.startMoisture.toFixed(0)}% → ${last.endMoisture.toFixed(0)}%`;
    $('#stEtaK').textContent = 'Finished'; $('#stEta').textContent = fmtTime(last.end); $('#stEtaS').textContent = ago(last.end);
  } else {
    $('#statCap').textContent = 'No sessions yet';
    ['stDur','stWater','stGain','stEta'].forEach(id => $('#'+id).textContent = '—');
    ['stDurS','stWaterS','stGainS','stEtaS'].forEach(id => $('#'+id).innerHTML = '&nbsp;');
  }

  // note
  let note;
  if(ses) note = ses.trigger === 'auto' ? `Automatic cycle running. It will stop at ${cfg.high}% or after ${cfg.maxRunMin} min.` : 'Manual watering. Turn the pump off at any time.';
  else if(safety) note = ls.reason === 'max_runtime' ? 'The pump was stopped by the max-run-time safety limit.' : 'The pump was stopped by overflow protection (soil nearly saturated).';
  else if(st.mode === 'auto' && st.cooldownRemainingSec > 0) note = `Cool-down active — next automatic check in ${fmtDur(st.cooldownRemainingSec)}.`;
  else if(st.mode === 'auto'){ const gap = st.moisture - cfg.low; note = gap <= 0 ? 'Below the start threshold — starting shortly.' : `Automatic mode armed — starts below ${cfg.low}% (${gap.toFixed(1)} points to go).`; }
  else note = 'Manual mode — the pump runs only when you switch it on.';
  $('#procNote').textContent = note;
}

async function loadData(){
  try{
    S.settings = await Api.getSettings(); S.draft = { ...S.settings };
    S.history = await Api.getHistory(24*3600e3); S.lastHist = S.history.length ? S.history[S.history.length-1].t : 0;
    S.sessions = await Api.getSessions(30); S.loaded = true;
  }catch(e){ S.loaded = false; }
  syncSettingsUI(); renderHistoryView(); render();
}

async function refreshSessions(){ try{ S.sessions = await Api.getSessions(30); }catch(e){} if(S.view==='history') renderHistoryView(); }

function onStatus(st){
  const first = S.firstStatus; S.firstStatus = false;
  S.status = st; S.online = true; S.connecting = false; S.fails = 0; S.lastOk = Date.now();
  if(!S.loaded) loadData();
  if(st.ts - S.lastHist >= 5000){
    S.history.push({ t:st.ts, v:st.moisture }); S.lastHist = st.ts;
    while(S.history.length && S.history[0].t < st.ts - 25*3600e3) S.history.shift();
  }
  const active = !!st.session;
  if(!first){
    if(active && !S.prevActive && st.session.trigger === 'auto') toast(`Automatic irrigation started — soil at ${st.moisture.toFixed(0)}%`, 'info');
    if(!active && S.prevActive){
      const ls = st.lastStop;
      if(ls && ls.reason === 'target_reached') toast(`Irrigation complete — target ${S.settings.high}% reached`, 'ok');
      else if(ls && ls.reason === 'max_runtime') toast('Pump stopped: maximum run time reached', 'warn');
      else if(ls && ls.reason === 'overwater') toast('Pump stopped: overflow protection triggered', 'warn');
      refreshSessions();
    }
  }
  S.prevActive = active;
  render();
}
function onFail(e){
  S.fails++; S.lastError = e.message;
  if(S.fails >= 2 || !S.status){ S.online = false; S.connecting = false; }
  render(); syncSettingsUI();
}
async function poll(token){
  if(token !== S.pollToken) return;
  try{ const st = await Api.getStatus(); if(token !== S.pollToken) return; const was = S.online; onStatus(st); if(!was) syncSettingsUI(); }

async function command(fn, okMsg){
  if(!S.online || S.busy) return;
  S.busy = true; $('#pumpSwitch').setAttribute('aria-busy','true');
  try{ const st = await fn(); if(st) onStatus(st); if(okMsg) toast(okMsg.text, okMsg.type); }
  catch(e){ toast('Command failed: ' + e.message, 'error'); }
  finally{ S.busy = false; $('#pumpSwitch').removeAttribute('aria-busy'); }
}
function togglePump(){
  if(!S.online){ toast('Device is offline — controls are disabled', 'warn'); return; }
  const st = S.status, next = !st.pump;
  command(async () => {
    if(st.mode === 'auto'){ await Api.setMode('manual'); toast('Switched to Manual mode to take control of the pump', 'info'); }
    return Api.setPump(next);
  }, { text: next ? 'Pump started' : 'Pump stopped', type: next ? 'ok' : 'info' });
}
